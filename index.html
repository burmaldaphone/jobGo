function renderVacancies(vacancies) { const vacanciesList = document.getElementById('vacanciesList'); if (vacancies.length === 0) { vacanciesList.innerHTML = `
Нет вакансий, соответствующих выбранным фильтрам.
`; return; } vacanciesList.innerHTML = vacancies.map(v => { // Цвет плашки в зависимости от площадки let sourceBadge = 'bg-gray-100 text-gray-600 border border-gray-200'; if (v.source === 'HeadHunter') sourceBadge = 'bg-red-50 text-red-600 border border-red-100'; if (v.source === 'Хабр Карьера') sourceBadge = 'bg-blue-50 text-blue-600 border border-blue-100'; if (v.source === 'Работа в России') sourceBadge = 'bg-emerald-50 text-emerald-700 border border-emerald-100'; return `

${v.title}

${v.source}
${v.company}
${v.salary}

${v.snippet}

Смотреть на ${v.source} ↗
`; }).join(''); }
